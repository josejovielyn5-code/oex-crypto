async function loadUsers() { const select = document.getElementById("userSelect"); const subtractSelect = document.getElementById("subtractUserSelect"); if (!select) { console.error("userSelect not found"); return; } if (!subtractSelect) { console.error("subtractUserSelect not found"); return; } const { data, error } = await supabaseClient .from("profiles") .select("id, display_name, email") .order("display_name"); if (error) { console.error(error); return; } // Clear Add Balance dropdown select.innerHTML = ''; // Clear Subtract Balance dropdown subtractSelect.innerHTML = ''; data.forEach(user => { const userText = (user.display_name || "OEX User") + " — " + (user.email || "No email"); // Add Balance dropdown const addOption = document.createElement("option"); addOption.value = user.id; addOption.textContent = userText; select.appendChild(addOption); // Subtract Balance dropdown const subtractOption = document.createElement("option"); subtractOption.value = user.id; subtractOption.textContent = userText; subtractSelect.appendChild(subtractOption); }); }
